=== c-tabela | Se a tabela rolar na horizontal
/* Quatro causas já vistas:
   1. falta  min-width: 0; width: 100%  no modificador da tela;
   2. texto longo sem truncagem  →  overflow: hidden; text-overflow: ellipsis; white-space: nowrap;
   3. imagem dentro de <td>      →  width: 100%; max-width: 60px; height: auto; aspect-ratio: 1; min-width: 0;
   4. rótulo de coluna de palavra única maior que a coluna → largura = rótulo + 32px, arredondada para múltiplo de 8.
   Regras fixas: padding de célula SEMPRE 8px 16px (denso: 8px 8px). Nunca 6, 7, 10 ou 12.
   Não há ordenação por clique no <th>: a ordenação é sempre pelo select da barra de resultados. */

=== Badges | Badge.tsx
/* Badge de status (pílula) e badge de apresentação (sigla 22×22). */
const APRESENTACAO: Record<string, string> = { N: 'Nominativa', M: 'Mista', F: 'Figurativa' }

export function ApresentacaoBadge({ sigla }: { sigla: string | null | undefined }) {
  const s = sigla?.trim().toUpperCase() || null
  return <span className={`badge-sigla${s ? ` badge-sigla--${s}` : ''}`} title={s ? APRESENTACAO[s] ?? s : undefined}>{s || '—'}</span>
}

export function StatusBadge({ tom, children }: { tom: 'success' | 'info' | 'warning' | 'danger'; children: React.ReactNode }) {
  return <span className={`badge badge--${tom}`}>{children}</span>
}

    <div className="kpi">
      <span className="kpi__rotulo">{rotulo}</span>
      <strong className="kpi__valor">{valor}</strong>
      {detalhe && <span className="kpi__detalhe">{detalhe}</span>}

  )
}

import type { InputHTMLAttributes, ReactNode } from 'react'

interface FieldProps extends InputHTMLAttributes<HTMLInputElement> {
  label: string
  /** 'stack' = rótulo em cima (.fg, formulários de detalhe). 'row' = rótulo ao lado (.mf-row, filtros). */
  layout?: 'stack' | 'row'
  error?: string
  hint?: ReactNode
}

export function Field({ label, layout = 'stack', error, hint, id, ...input }: FieldProps) {
  const campo = id ?? `f-${label.toLowerCase().replace(/\W+/g, '-')}`
  const entrada = (
    <>
      <input id={campo} aria-invalid={error ? true : undefined} aria-describedby={error ? `${campo}-erro` : undefined} {...input} />
      {error && <div id={`${campo}-erro`} className="field-error">{error}</div>}
      {hint && !error && <div className="field-hint">{hint}</div>}
    </>
  )
  return layout === 'row' ? (
    <div className="field-row">
      <label className="field-row__lbl" htmlFor={campo}>{label}:</label>
      <div className="field-row__val">{entrada}</div>
    </div>
  ) : (
    <div className="fg">
      <label htmlFor={campo}>{label}</label>
      {entrada}
    </div>
  )
}

}

/* Autocomplete feito à mão, sem lib. Três pegadinhas obrigatórias:
   1) escolher com onMouseDown (onClick chega depois do blur e perde o item);
   2) onBlur com atraso (setTimeout 150) para o mousedown acontecer antes de fechar;
   3) interceptar Enter para não submeter o formulário. */
import { useState } from 'react'

interface Item { id: string; nome: string }

export function Autocomplete({ itens, onEscolher, placeholder }: { itens: Item[]; onEscolher: (i: Item) => void; placeholder?: string }) {
  const [texto, setTexto] = useState('')
  const [aberto, setAberto] = useState(false)
  const filtrados = itens.filter(i => i.nome.toLowerCase().includes(texto.toLowerCase()))

  return (
    <div style={{ position: 'relative', flex: 1 }}>                       {/* wrapper precisa de position: relative */}
      <input
        value={texto} placeholder={placeholder}
        onChange={e => { setTexto(e.target.value); setAberto(true) }}
        onFocus={() => setAberto(true)}
        onBlur={() => setTimeout(() => setAberto(false), 150)}
        onKeyDown={e => { if (e.key === 'Enter' && aberto) e.preventDefault() }}
      />
      {aberto && (
        <div className="ac-lista" role="listbox">
          {filtrados.map(i => (
            <div key={i.id} role="option" className="ac-item" onMouseDown={() => { onEscolher(i); setTexto(i.nome); setAberto(false) }}>{i.nome}</div>
          ))}
          {filtrados.length === 0 && <div className="ac-msg">Nenhum resultado</div>}
        </div>
      )}
    </div>
  )
}

/* Botão que abre painel de checkboxes com "Todos" no topo. O painel vai para PORTAL porque o card de
   filtros tem overflow e cortaria um painel absoluto; abre para cima se faltar espaço embaixo. */
import { useEffect, useRef, useState } from 'react'
import { createPortal } from 'react-dom'
import { CaretDown } from '@phosphor-icons/react'

const ALTURA_MAXIMA = 260

function usePosicaoPopover(aberto: boolean, gatilho: React.RefObject<HTMLElement | null>) {
  const [pos, setPos] = useState<{ top: number; left: number; width: number; bottom: number; paraCima: boolean; maxHeight: number } | null>(null)
  useEffect(() => {
    if (!aberto) return
    function atualizar() {
      const r = gatilho.current?.getBoundingClientRect()
      if (!r) return
      const abaixo = window.innerHeight - r.bottom - 12, acima = r.top - 12
      const paraCima = abaixo < 180 && acima > abaixo
      setPos({ top: r.bottom + 4, left: r.left, width: r.width, bottom: window.innerHeight - r.top + 4, paraCima,
        maxHeight: Math.max(120, Math.min(ALTURA_MAXIMA, paraCima ? acima : abaixo)) })
    }
    atualizar()
    window.addEventListener('scroll', atualizar, true)
    window.addEventListener('resize', atualizar)
    return () => { window.removeEventListener('scroll', atualizar, true); window.removeEventListener('resize', atualizar) }
  }, [aberto, gatilho])
  return pos
}

export interface Opcao { chave: string; rotulo: string }

export function PopoverMultiSelect({ opcoes, selecionados, singular, plural, zIndex, onToggle, onToggleTodos }: {
  opcoes: Opcao[]; selecionados: string[]; singular: string; plural: string
  zIndex?: number                      // dentro de drawer (z 8001+) o padrão (40) ficaria por baixo
  onToggle: (chave: string) => void; onToggleTodos: () => void
}) {
  const [aberto, setAberto] = useState(false)
  const caixa = useRef<HTMLDivElement>(null), botao = useRef<HTMLButtonElement>(null), painel = useRef<HTMLDivElement>(null)
  const pos = usePosicaoPopover(aberto, botao)
  const todos = opcoes.length > 0 && selecionados.length === opcoes.length

  useEffect(() => {
    if (!aberto) return
    const fora = (e: MouseEvent) => { const a = e.target as Node; if (!caixa.current?.contains(a) && !painel.current?.contains(a)) setAberto(false) }
    document.addEventListener('mousedown', fora)
    return () => document.removeEventListener('mousedown', fora)
  }, [aberto])

  const rotulo = selecionados.length === 0 ? 'Selecione' : `${selecionados.length.toLocaleString('pt-BR')} ${selecionados.length === 1 ? singular : plural}`

  return (
    <div ref={caixa} style={{ position: 'relative', flex: 1 }}>
      <button ref={botao} type="button" className="ms-btn" onClick={() => setAberto(v => !v)}>
        <span className="ms-btn__txt">{rotulo}</span><CaretDown size={12} weight="bold" />
      </button>
      {aberto && pos && createPortal(
        <div ref={painel} className="ms-painel"
          style={{ position: 'fixed', left: pos.left, width: pos.width, zIndex, maxHeight: pos.maxHeight,
            ...(pos.paraCima ? { bottom: pos.bottom } : { top: pos.top }) }}>
          <div className="ms-item ms-item--todos" onClick={onToggleTodos}><input type="checkbox" checked={todos} readOnly /> Todos</div>
          {opcoes.map(o => (
            <div key={o.chave} className="ms-item" onClick={() => onToggle(o.chave)}>
              <input type="checkbox" checked={selecionados.includes(o.chave)} readOnly /> {o.rotulo}
            </div>
          ))}
        </div>, document.body)}
    </div>
  )
}

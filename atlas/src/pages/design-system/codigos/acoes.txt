import type { ButtonHTMLAttributes, ReactNode } from 'react'
import { ArrowClockwise } from '@phosphor-icons/react'
import './button.css'

export type ButtonTone = 'primary' | 'secondary' | 'neutral' | 'danger' | 'success'   // Figma: Type
export type ButtonVariant = 'filled' | 'outline' | 'text'                              // Figma: Style
export type ButtonSize = 'small' | 'medium' | 'large'

interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  tone?: ButtonTone
  variant?: ButtonVariant
  size?: ButtonSize
  /** Ícone Phosphor antes do rótulo (use size 16 em small/medium, 20 em large). */
  icon?: ReactNode
  /** Só ícone, sem rótulo: o botão vira quadrado. Passe aria-label. */
  iconOnly?: boolean
  /** Mostra só o spinner e bloqueia o clique; a altura é mantida. */
  loading?: boolean
  children?: ReactNode
}

export function Button({
  tone = 'primary', variant = 'filled', size = 'small',
  icon, iconOnly = false, loading = false, className = '', children, disabled, type = 'button', ...rest
}: ButtonProps) {
  return (
    <button
      type={type}
      className={`btn btn--${tone} btn--${variant} btn--${size}${iconOnly ? ' btn--icon' : ''}${loading ? ' is-loading' : ''} ${className}`}
      disabled={disabled || loading}
      aria-busy={loading || undefined}
      {...rest}
    >
      {loading ? <ArrowClockwise className="btn__spin" size={16} /> : <>{icon}{!iconOnly && children}</>}
    </button>
  )
}

import { useEffect, useRef, useState } from 'react'
import { CaretDown, CaretRight, DownloadSimple, Trash } from '@phosphor-icons/react'

function Item({ icone, rotulo, perigo, onClick }: { icone?: React.ReactNode; rotulo: string; perigo?: boolean; onClick: () => void }) {
  return (
    <button type="button" role="menuitem" className={`pac-item${perigo ? ' pac-item-perigo' : ''}`} onClick={onClick}>
      {icone}{rotulo}
    </button>
  )
}

export function MenuDeAcoes({ selecionados, onAcao }: { selecionados: number; onAcao: (a: string) => void }) {
  const [aberto, setAberto] = useState(false)
  const [submenu, setSubmenu] = useState(false)
  const wrapRef = useRef<HTMLDivElement>(null)

  /* fecha por mousedown fora do menu */
  useEffect(() => {
    if (!aberto) return
    function fora(e: MouseEvent) {
      if (!wrapRef.current?.contains(e.target as Node)) { setAberto(false); setSubmenu(false) }
    }
    document.addEventListener('mousedown', fora)
    return () => document.removeEventListener('mousedown', fora)
  }, [aberto])

  const executar = (a: string) => { setAberto(false); setSubmenu(false); onAcao(a) }

  return (
    <div className="pac-wrap" ref={wrapRef}>
      <button type="button" className="pac-btn" aria-haspopup="menu" aria-expanded={aberto} onClick={() => setAberto(v => !v)}>
        Ações{selecionados > 0 ? ` (${selecionados})` : ''}
        <CaretDown size={12} className={aberto ? 'pac-chev-aberto' : undefined} />
      </button>
      {aberto && (
        <div className="pac-menu" role="menu">
          <Item icone={<DownloadSimple size={15} />} rotulo="Exportar XLS" onClick={() => executar('xls')} />
          <div className="pac-sep" />
          <div className="pac-sub" onMouseEnter={() => setSubmenu(true)}>
            <button type="button" className="pac-item" onClick={() => setSubmenu(v => !v)}>
              Mais opções <CaretRight size={12} className="pac-sub-seta" />
            </button>
            {submenu && <div className="pac-submenu">{/* itens do submenu */}</div>}
          </div>
          <div className="pac-sep" />
          <Item icone={<Trash size={15} />} rotulo="Excluir" perigo onClick={() => executar('excluir')} />
        </div>
      )}
    </div>
  )
}

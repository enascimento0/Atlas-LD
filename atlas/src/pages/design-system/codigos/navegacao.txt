=== Tabs | Tabs.tsx
interface Aba { id: string; rotulo: string }

export function Tabs({ abas, ativa, onChange }: { abas: Aba[]; ativa: string; onChange: (id: string) => void }) {
  return (
    <div className="tabs" role="tablist">
      {abas.map(a => (
        <button key={a.id} type="button" role="tab" aria-selected={a.id === ativa}
          className={`tab${a.id === ativa ? ' is-ativa' : ''}`} onClick={() => onChange(a.id)}>
          {a.rotulo}
        </button>
      ))}
    </div>
  )
}

=== Pagination | Paginacao.tsx
import { CaretLeft, CaretRight } from '@phosphor-icons/react'

export const TAMANHOS_PAGINA = [20, 50, 100]          // padrão 20

/** Até 7 páginas mostra todas; acima disso → 1 … [atual−2 … atual+2] … total */
export function gerarPaginas(atual: number, total: number): (number | '...')[] {
  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1)
  const p: (number | '...')[] = [1]
  if (atual > 3) p.push('...')
  for (let n = Math.max(2, atual - 2); n <= Math.min(total - 1, atual + 2); n++) p.push(n)
  if (atual < total - 2) p.push('...')
  p.push(total)
  return p
}

export function Paginacao({ pagina, tamanho, total, rotulo = 'registros', onPagina, onTamanho }: {
  pagina: number; tamanho: number; total: number; rotulo?: string
  onPagina: (p: number) => void; onTamanho: (t: number) => void
}) {
  const totalPaginas = Math.max(1, Math.ceil(total / tamanho))
  const inicio = total === 0 ? 0 : (pagina - 1) * tamanho + 1
  const fim = Math.min(pagina * tamanho, total)
  const n = (v: number) => v.toLocaleString('pt-BR')

  return (
    <div className="pag">
      <span className="pag__info">Exibindo <strong>{n(inicio)}–{n(fim)}</strong> de <strong>{n(total)}</strong> {rotulo}</span>
      <div className="pag__controles">
        {/* trocar o tamanho SEMPRE volta para a página 1 (idem ao trocar a ordenação) */}
        <select className="pag__tam" value={tamanho} onChange={e => { onTamanho(Number(e.target.value)); onPagina(1) }}>
          {TAMANHOS_PAGINA.map(t => <option key={t} value={t}>{t} por página</option>)}
        </select>
        <div className="pag__btns">
          <button className="pag__btn" disabled={pagina <= 1} onClick={() => onPagina(pagina - 1)} title="Página anterior"><CaretLeft size={12} /></button>
          {gerarPaginas(pagina, totalPaginas).map((p, i) => p === '...'
            ? <span key={`e${i}`} className="pag__reticencias">…</span>
            : <button key={p} className={`pag__btn${p === pagina ? ' is-ativa' : ''}`} onClick={() => onPagina(p)}>{p}</button>)}
          <button className="pag__btn" disabled={pagina >= totalPaginas} onClick={() => onPagina(pagina + 1)} title="Próxima página"><CaretRight size={12} /></button>
        </div>
      </div>
    </div>
  )
}

/* Bloqueia a rota se a funcionalidade não está no plano do cliente.
   É só UX — a segurança real é o 403 do backend. */
export function FuncionalidadeRoute({ code, nome, children }: { code: string; nome: string; children: ReactNode }) {
  const { data } = useFuncionalidadesDoTenant()
  if (!data) return <>{children}</>                              // carregando: não pisca "Não contratado"
  const habilitada = data.produtos.flatMap(p => p.modules).flatMap(m => m.functionalities).some(f => f.code === code && f.habilitada)
  return habilitada ? <>{children}</> : <NaoContratadoPage nomeFuncionalidade={nome} />
}

/* Tema = classe no <body> (muda só --c-support* e --c-sidebar-bg).
   Modo escuro = classe no <html>. As duas são independentes e se combinam. */
import { useEffect, useState } from 'react'

export interface AppTheme { id: string; label: string; support: string; sidebarBg: string }

export const APP_THEMES: AppTheme[] = [
  { id: 'padrao',  label: 'Padrão',  support: '#0059AD', sidebarBg: '#001A33' },
  { id: 'violeta', label: 'Violeta', support: '#7C3AED', sidebarBg: '#241B3D' },
  { id: 'ambar',   label: 'Âmbar',   support: '#D97706', sidebarBg: '#3A2607' },
  { id: 'teal',    label: 'Teal',    support: '#0D9488', sidebarBg: '#062A28' },
  { id: 'rosa',    label: 'Rosa',    support: '#DB2777', sidebarBg: '#3D1225' },
  { id: 'cinza',   label: 'Cinza',   support: '#475569', sidebarBg: '#1E293B' },
  { id: 'marinho', label: 'Marinho', support: '#001A33', sidebarBg: '#000D1A' },
]

const THEME_KEY = 'userTheme'
const DARK_KEY = 'darkMode'

function temaSalvo(): string {
  const s = localStorage.getItem(THEME_KEY)
  return s && APP_THEMES.some(t => t.id === s) ? s : 'padrao'
}
function aplicarTema(id: string) {
  for (const t of APP_THEMES) document.body.classList.remove(`theme-${t.id}`)
  if (id !== 'padrao') document.body.classList.add(`theme-${id}`)      // 'padrao' = sem classe
}

export function useAppTheme() {
  const [theme, setTheme] = useState<string>(temaSalvo)
  const [dark, setDark] = useState<boolean>(() => localStorage.getItem(DARK_KEY) === '1')
  useEffect(() => { aplicarTema(theme); localStorage.setItem(THEME_KEY, theme) }, [theme])
  useEffect(() => { document.documentElement.classList.toggle('dark-mode', dark); localStorage.setItem(DARK_KEY, dark ? '1' : '0') }, [dark])
  return { theme, setTheme, themes: APP_THEMES, dark, setDark }
}

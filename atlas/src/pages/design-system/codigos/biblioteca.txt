/** Um item aberto por vez (multiplo = vários). */
export function Accordion({ itens, multiplo = false }: { itens: Item[]; multiplo?: boolean }) {
  const [abertos, setAbertos] = useState<string[]>([])
  const alternar = (id: string) => setAbertos(a => a.includes(id) ? a.filter(x => x !== id) : multiplo ? [...a, id] : [id])
  return (
    <div className="acc">
      {itens.map(i => {
        const aberto = abertos.includes(i.id)
        return (
          <div key={i.id} className={`acc__item${aberto ? ' is-aberto' : ''}`}>
            <button type="button" className="acc__btn" aria-expanded={aberto} onClick={() => alternar(i.id)}>
              <span>{i.titulo}</span><CaretDown size={14} className="acc__seta" />
            </button>
            {aberto && <div className="acc__corpo">{i.conteudo}</div>}
          </div>
        )
      })}
    </div>
  )
}

=== Dropdown | Dropdown.tsx
import { useEffect, useRef, useState } from 'react'
import { CaretDown, Check } from '@phosphor-icons/react'

interface Opcao { valor: string; rotulo: string }

/** Lista suspensa estilizada. Fecha por mousedown fora e com Esc. Para poucas opções, prefira <select>. */
export function Dropdown({ opcoes, valor, onChange, placeholder = 'Selecione' }: { opcoes: Opcao[]; valor: string; onChange: (v: string) => void; placeholder?: string }) {
  const [aberto, setAberto] = useState(false)
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    if (!aberto) return
    const fora = (e: MouseEvent) => { if (!ref.current?.contains(e.target as Node)) setAberto(false) }
    const tecla = (e: KeyboardEvent) => { if (e.key === 'Escape') setAberto(false) }
    document.addEventListener('mousedown', fora); document.addEventListener('keydown', tecla)
    return () => { document.removeEventListener('mousedown', fora); document.removeEventListener('keydown', tecla) }
  }, [aberto])
  const atual = opcoes.find(o => o.valor === valor)
  return (
    <div className="dd" ref={ref}>
      <button type="button" className="dd__btn" aria-haspopup="listbox" aria-expanded={aberto} onClick={() => setAberto(v => !v)}>
        <span className={atual ? '' : 'dd__ph'}>{atual?.rotulo ?? placeholder}</span><CaretDown size={12} weight="bold" />
      </button>
      {aberto && (
        <ul className="dd__lista" role="listbox">
          {opcoes.map(o => (
            <li key={o.valor} role="option" aria-selected={o.valor === valor} className={`dd__item${o.valor === valor ? ' is-sel' : ''}`} onClick={() => { onChange(o.valor); setAberto(false) }}>
              {o.rotulo}{o.valor === valor && <Check size={14} weight="bold" />}
            </li>
          ))}
        </ul>
      )}
    </div>
  )
}

/** Campo numérico: aceita só dígitos, mostra separador de milhar pt-BR no blur. */
import { useState } from 'react'

export function NumberField({ valor, onChange, min, max }: { valor: number | null; onChange: (v: number | null) => void; min?: number; max?: number }) {
  const [texto, setTexto] = useState(valor == null ? '' : String(valor))
  const [foco, setFoco] = useState(false)
  const mostrar = foco || valor == null ? texto : valor.toLocaleString('pt-BR')
  return (
    <input
      className="num" inputMode="numeric" value={mostrar}
      onFocus={() => { setFoco(true); setTexto(valor == null ? '' : String(valor)) }}
      onBlur={() => setFoco(false)}
      onChange={e => {
        const limpo = e.target.value.replace(/\D/g, '')
        setTexto(limpo)
        if (limpo === '') return onChange(null)
        let n = Number(limpo)
        if (min != null) n = Math.max(min, n)
        if (max != null) n = Math.min(max, n)
        onChange(n)
      }}
    />
  )
}

/** Assistente em etapas: Voltar / Próximo / Concluir. Próximo fica desabilitado enquanto o passo não for válido. */
export function Wizard({ passos, onConcluir }: { passos: Passo[]; onConcluir: () => void }) {

      <div className="wz__rodape">
        <Button variant="outline" tone="neutral" disabled={i === 0} onClick={() => setI(i - 1)}>Voltar</Button>
        <Button disabled={passo.valido === false} onClick={() => ultimo ? onConcluir() : setI(i + 1)}>{ultimo ? 'Concluir' : 'Próximo'}</Button>
      </div>
    </div>
  )
}

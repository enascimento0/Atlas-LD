=== Toast | ToastProvider.tsx
/* Dois tipos num só provider, em PORTAL no canto inferior direito (bottom/right 28px, z-index 9500).
   - Toast simples (uma linha):  showToast(msg, variant?, action?, duracaoMs?)
   - Notificação (título+texto): showNotificacao({ tipo, titulo, mensagem?, acao?, duracaoMs?, chave? })
   Padrão: 4s (8s com ação "Desfazer"); notificação 10s (duracaoMs: null = fica até fechar); máx. 4 empilhadas. */
import { createContext, useCallback, useContext, useMemo, useRef, useState, type ReactNode } from 'react'
import { createPortal } from 'react-dom'
import { CheckCircle, Info, Megaphone, Siren, WarningCircle, X, type Icon } from '@phosphor-icons/react'

export type ToastVariant = 'success' | 'error' | 'info' | 'warning'
export interface ToastAction { label: string; onClick: () => void }
interface ToastItem { id: number; message: string; variant: ToastVariant; action?: ToastAction }

export type TipoNotificacao = 'prazo' | 'alerta' | 'info' | 'sucesso' | 'promocao'
export interface Notificacao { tipo: TipoNotificacao; titulo: string; mensagem?: string; acao?: ToastAction; duracaoMs?: number | null; chave?: string }
interface NotificacaoItem extends Notificacao { id: number }

interface Ctx {
  showToast: (m: string, v?: ToastVariant, a?: ToastAction, duracaoMs?: number) => void
  showNotificacao: (n: Notificacao) => void
}
const ToastContext = createContext<Ctx | null>(null)

const ICONE_TOAST: Record<ToastVariant, ReactNode> = {
  success: <CheckCircle size={16} weight="fill" />, error: <WarningCircle size={16} weight="fill" />,
  info: <Info size={16} weight="fill" />, warning: <WarningCircle size={16} weight="fill" />,
}
const ICONE_NOTIF: Record<TipoNotificacao, Icon> = { prazo: WarningCircle, alerta: Siren, info: Info, sucesso: CheckCircle, promocao: Megaphone }

export function ToastProvider({ children }: { children: ReactNode }) {
  const [toasts, setToasts] = useState<ToastItem[]>([])
  const [notifs, setNotifs] = useState<NotificacaoItem[]>([])
  const seq = useRef(0)

  const fecharNotif = useCallback((id: number) => setNotifs(p => p.filter(n => n.id !== id)), [])
  const fecharToast = useCallback((id: number) => setToasts(p => p.filter(t => t.id !== id)), [])

  const showNotificacao = useCallback((n: Notificacao) => {
    const id = ++seq.current
    setNotifs(p => (n.chave && p.some(x => x.chave === n.chave)) ? p : [...p, { ...n, id }].slice(-4))   // mesma chave = não repete
    const dur = n.duracaoMs === undefined ? 10000 : n.duracaoMs
    if (dur !== null) window.setTimeout(() => fecharNotif(id), dur)
  }, [fecharNotif])

  const showToast = useCallback((message: string, variant: ToastVariant = 'info', action?: ToastAction, duracaoMs?: number) => {
    const id = ++seq.current
    setToasts(p => [...p, { id, message, variant, action }])
    window.setTimeout(() => fecharToast(id), duracaoMs ?? (action ? 8000 : 4000))
  }, [fecharToast])

  const value = useMemo(() => ({ showToast, showNotificacao }), [showToast, showNotificacao])

  return (
    <ToastContext.Provider value={value}>
      {children}
      {createPortal(
        <div className="toast-pilha">
          {notifs.map(n => { const I = ICONE_NOTIF[n.tipo]; return (
            <div key={n.id} className={`toast-notif toast-notif--${n.tipo}`} role={n.tipo === 'alerta' || n.tipo === 'prazo' ? 'alert' : 'status'}>
              <span className="toast-notif__icone"><I size={18} weight="fill" /></span>
              <div className="toast-notif__textos">
                <span className="toast-notif__titulo">{n.titulo}</span>
                {n.mensagem && <span className="toast-notif__msg">{n.mensagem}</span>}
                {n.acao && <button type="button" className="toast-notif__acao" onClick={() => { n.acao?.onClick(); fecharNotif(n.id) }}>{n.acao.label}</button>}
              </div>
              <button type="button" className="toast-notif__fechar" onClick={() => fecharNotif(n.id)} aria-label="Fechar notificação"><X size={14} /></button>
            </div>
          ) })}
          {toasts.map(t => (
            <div key={t.id} role="status" className={`toast toast--${t.variant}`}>
              <span style={{ display: 'flex', flexShrink: 0 }}>{ICONE_TOAST[t.variant]}</span>
              <span>{t.message}</span>
              {t.action && <button type="button" className="toast__acao" onClick={() => { t.action?.onClick(); fecharToast(t.id) }}>{t.action.label}</button>}
              <button type="button" className="toast__fechar" onClick={() => fecharToast(t.id)} aria-label="Fechar notificação"><X size={13} weight="bold" /></button>
            </div>
          ))}
        </div>, document.body)}
    </ToastContext.Provider>
  )
}

export function useToast() {
  const ctx = useContext(ToastContext)
  if (!ctx) throw new Error('useToast deve ser usado dentro de <ToastProvider>')
  return ctx
}

import { useRef, useState } from 'react'
import { CornersIn, CornersOut } from '@phosphor-icons/react'

export function BotaoTelaCheia() {
  const ref = useRef<HTMLButtonElement>(null)
  const [cheia, setCheia] = useState(false)
  function alternar() {
    const drawer = ref.current?.closest('.drw')
    if (!drawer) return
    const proximo = !drawer.classList.contains('drawer-tela-cheia')
    drawer.classList.toggle('drawer-tela-cheia', proximo)
    setCheia(proximo)
  }
  const titulo = cheia ? 'Sair da tela cheia' : 'Tela cheia'
  return (
    <button ref={ref} type="button" className="drw__x" onClick={alternar} title={titulo} aria-label={titulo}>
      {cheia ? <CornersIn size={16} /> : <CornersOut size={16} />}
    </button>
  )
}

/* Quatro variantes — o componente diz POR QUE está vazio e oferece o próximo passo.
   inicio        tela aberta, ainda não pesquisou — texto começa com "Comece…", SEM botões

import type { ReactNode } from 'react'
import { ListMagnifyingGlass, MagnifyingGlass, Tray, WarningCircle, X, type Icon } from '@phosphor-icons/react'

export type Variante = 'inicio' | 'sem-resultado' | 'sem-cadastro' | 'erro'
export interface FiltroAtivo { chave: string; rotulo: string; valor: string }
export interface AcaoVazio { rotulo: string; onClick: () => void; tipo?: 'primaria' | 'secundaria' }

const ICONES: Record<Variante, { icone: Icon; cor: string }> = {
  inicio: { icone: ListMagnifyingGlass, cor: 'var(--c-support)' },
  'sem-resultado': { icone: MagnifyingGlass, cor: 'var(--c-text-muted)' },
  'sem-cadastro': { icone: Tray, cor: 'var(--c-support)' },
  erro: { icone: WarningCircle, cor: 'var(--c-danger)' },
}

export function EstadoVazio({ variante, titulo, descricao, filtrosAtivos, onRemoverFiltro, acoes }: {
  variante: Variante; titulo: string; descricao?: ReactNode
  filtrosAtivos?: FiltroAtivo[]; onRemoverFiltro?: (chave: string) => void; acoes?: AcaoVazio[]
}) {
  const { icone: I, cor } = ICONES[variante]
  return (
    <div className={`vazio vazio--${variante}`} role={variante === 'erro' ? 'alert' : 'status'}>
      <span className={`vazio__icone vazio__icone--${variante}`}><I size={32} color={cor} /></span>
      <strong className="vazio__titulo">{titulo}</strong>
      {descricao && <p className="vazio__desc">{descricao}</p>}
      {filtrosAtivos?.length ? (
        <div className="vazio__filtros">
          {filtrosAtivos.map(f => (
            <span key={f.chave} className="vazio__chip">
              <span style={{ color: 'var(--c-neutral)' }}>{f.rotulo}:</span> {f.valor}
              {onRemoverFiltro && <button type="button" onClick={() => onRemoverFiltro(f.chave)} aria-label={`Remover o filtro ${f.rotulo} e pesquisar de novo`}><X size={10} weight="bold" /></button>}
            </span>
          ))}
        </div>
      ) : null}
      {acoes?.length ? (
        <div className="vazio__acoes">
          {acoes.map(a => <Button key={a.rotulo} variant={a.tipo === 'primaria' ? 'filled' : 'outline'} tone={a.tipo === 'primaria' ? 'primary' : 'neutral'} onClick={a.onClick}>{a.rotulo}</Button>)}
        </div>
      ) : null}
    </div>
  )
}

import { useState } from 'react'
import { Question, X } from '@phosphor-icons/react'

export function BotaoDeAjuda({ topicos }: { topicos: { id: string; titulo: string; conteudo: React.ReactNode }[] }) {
  const [aberto, setAberto] = useState(false)
  const [topico, setTopico] = useState<string | null>(null)
  return (
    <>
      {!aberto && <button type="button" className="ajuda-fab" onClick={() => setAberto(true)} aria-label="Ajuda desta tela"><Question size={22} weight="bold" /></button>}
      {aberto && (
        <div className="ajuda-painel" role="dialog" aria-label="Ajuda desta tela">
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', padding: '14px 16px' }}>
            <strong style={{ fontSize: 14 }}>Ajuda desta tela</strong>
            <button type="button" className="drw__x" onClick={() => setAberto(false)} aria-label="Fechar"><X size={16} weight="bold" /></button>
          </div>
          <div style={{ display: 'flex', flexDirection: 'column', gap: 8, padding: '0 16px 16px', overflowY: 'auto', maxHeight: '60vh' }}>
            {topicos.map(t => (
              <div key={t.id} className={`ajuda-topico${topico === t.id ? ' is-aberto' : ''}`}>
                <button type="button" onClick={() => setTopico(topico === t.id ? null : t.id)}
                  style={{ width: '100%', padding: '10px 12px', border: 'none', background: 'none', textAlign: 'left', fontSize: 13, fontWeight: 600, color: 'var(--c-text)', cursor: 'pointer' }}>{t.titulo}</button>
                {topico === t.id && <div style={{ padding: '0 12px 12px', fontSize: 12.5, color: 'var(--c-neutral)' }}>{t.conteudo}</div>}
              </div>
            ))}
          </div>
        </div>
      )}
    </>
  )
}

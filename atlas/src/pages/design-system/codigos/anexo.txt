=== Date picker | Comportamento.md
Calendário próprio para datas, em portal (document.body). Dois componentes: SeletorDePeriodo (De → Até) e SeletorDeData (data única).
Os dois são o <input type="date"> nativo (aceita digitação; o ícone nativo some e fica o CalendarBlank) + um calendário próprio. Clicar ou focar abre o painel.

                 | Período (De → Até)                                      | Data única
-----------------|---------------------------------------------------------|----------------------------------------------
Meses exibidos   | 2 (1 se window.innerWidth < 720)                        | 1
Clique           | 1º clique = início, 2º = fim (antes do início, recomeça)| 1 clique grava e fecha
Destaque         | extremos em --c-support + faixa 12% entre eles          | só o dia escolhido (sem faixa)
Dica no hover    | tooltip escuro "N dias" sobre o dia (escolhendo o fim)  | não tem
Rodapé           | resumo "dd/mm/aaaa → dd/mm/aaaa · N dias" + Cancelar / Confirmar | data por extenso + Limpar / Hoje
Aplica quando    | clica em Confirmar (precisa De e Até)                   | ao clicar no dia, Limpar ou Hoje
Fecha            | Esc, Cancelar ou clique fora (descarta o rascunho)      | Esc ou clique fora (não altera)
Mês inicial      | mês anterior ao do "Até" (fica à direita) ou o do "De"  | mês do valor, ou hoje

Regras comuns: datas em ISO AAAA-MM-DD ("" = vazio); min/max desabilitam dias fora do limite; hoje aparece em negrito + sublinhado;
painel em position: fixed abaixo do campo, sem passar da borda direita da janela; z-index 9400 (acima de drawers e modais).
Dependências: React 19, @phosphor-icons/react, alias @/ → src/. Os tokens (--c-*, --dk-*) vêm do tokens.css do Design System.

=== Date picker | SeletorDePeriodo.tsx
import { useEffect, useRef, useState } from 'react'
import { createPortal } from 'react-dom'
import { CalendarBlank, CaretLeft, CaretRight, Check, X } from '@phosphor-icons/react'

// Período De → Até: os dois campos de data de sempre (aceitam digitação), mas o clique abre um
// calendário de dois meses em que o primeiro clique marca o início e o segundo o fim — o intervalo
// fica destacado e, ao escolher o fim, a dica mostra quantos dias o período tem. "Confirmar"
// aplica; Esc, Cancelar ou clique fora descartam. Datas em ISO (AAAA-MM-DD); "" = vazio.

export interface Periodo { de: string; ate: string }

export const MESES = ['Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho', 'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro']
const DIAS_DA_SEMANA = ['Dom', 'Seg', 'Ter', 'Qua', 'Qui', 'Sex', 'Sáb']
// Largura do painel (2 meses de 7 × 36px + vão + padding) — para não sair da tela à direita.
const LARGURA_DOIS_MESES = 2 * 7 * 36 + 32 + 40
export const LARGURA_UM_MES = 7 * 36 + 40

export function iso(ano: number, mes: number, dia: number): string {
  return `${ano}-${String(mes + 1).padStart(2, '0')}-${String(dia).padStart(2, '0')}`
}
export function hojeIso(): string {
  const d = new Date()
  return iso(d.getFullYear(), d.getMonth(), d.getDate())
}
export function formatar(valor: string): string {
  const [a, m, d] = valor.split('-')
  return `${d}/${m}/${a}`
}
function diasEntre(de: string, ate: string): number {
  const [a1, m1, d1] = de.split('-').map(Number)
  const [a2, m2, d2] = ate.split('-').map(Number)
  return Math.round((new Date(a2, m2 - 1, d2).getTime() - new Date(a1, m1 - 1, d1).getTime()) / 86_400_000) + 1
}
function plural(n: number): string {
  return `${n} ${n === 1 ? 'dia' : 'dias'}`
}

export type MesVisivel = { ano: number; mes: number }

export function somarMeses({ ano, mes }: MesVisivel, n: number): MesVisivel {
  const d = new Date(ano, mes + n, 1)
  return { ano: d.getFullYear(), mes: d.getMonth() }
}

/** Mês inicial da esquerda: o mês anterior ao do "Até" (o "Até" fica à direita), ou o do "De". */
function mesInicial(p: Periodo, meses: number): MesVisivel {
  const base = p.ate || p.de || hojeIso()
  const [a, m] = base.split('-').map(Number)
  return somarMeses({ ano: a, mes: m - 1 }, meses === 2 && p.ate ? -1 : 0)
}

export function MesDoCalendario({ visivel, rascunho, previa, onEscolher, onPassar, min, max }: {
  visivel: MesVisivel
  rascunho: Periodo
  /** Fim provisório enquanto o mouse passa (escolhendo o fim). */
  previa: string | null
  onEscolher: (dia: string) => void
  onPassar: (dia: string | null) => void
  /** Limites (ISO): os dias fora deles ficam desabilitados. */
  min?: string
  max?: string
}) {
  const { ano, mes } = visivel
  const primeiroDiaDaSemana = new Date(ano, mes, 1).getDay()
  const totalDeDias = new Date(ano, mes + 1, 0).getDate()
  const hoje = hojeIso()
  // Escolhendo o fim, o dia sob o mouse vale como fim provisório (mostra a faixa e a dica).
  const fim = (previa && previa >= rascunho.de ? previa : '') || rascunho.ate
  const inicio = rascunho.de

  return (
    <div>
      <div className="mf-periodo-mes-tit">{MESES[mes]} de {ano}</div>
      <div className="mf-periodo-grade" onMouseLeave={() => onPassar(null)}>
        {DIAS_DA_SEMANA.map(d => <span key={d} className="mf-periodo-sem">{d}</span>)}
        {Array.from({ length: primeiroDiaDaSemana }, (_, i) => <span key={`v${i}`} />)}
        {Array.from({ length: totalDeDias }, (_, i) => {
          const dia = iso(ano, mes, i + 1)
          const coluna = (primeiroDiaDaSemana + i) % 7
          const ehInicio = dia === inicio
          const ehFim = !!fim && dia === fim && fim >= inicio
          const noIntervalo = !!inicio && !!fim && dia > inicio && dia < fim
          const classes = ['mf-periodo-dia']
          if (ehInicio || ehFim) classes.push('ponta')
          if (noIntervalo) classes.push('no-intervalo')
          // Faixa do intervalo arredondada no começo e no fim de cada semana.
          if (noIntervalo && (coluna === 0 || i === 0)) classes.push('borda-esq')
          if (noIntervalo && (coluna === 6 || i === totalDeDias - 1)) classes.push('borda-dir')
          if (dia === hoje) classes.push('hoje')
          const mostrarDica = !!inicio && previa === dia && dia >= inicio
          const foraDoLimite = (!!min && dia < min) || (!!max && dia > max)
          return (
            <button
              key={dia} type="button" className={classes.join(' ')} disabled={foraDoLimite}
              onClick={() => onEscolher(dia)}
              onMouseEnter={() => onPassar(dia)}
              aria-label={formatar(dia)} aria-pressed={ehInicio || ehFim}
            >
              {i + 1}
              {mostrarDica && <span className="mf-periodo-dica">{plural(diasEntre(inicio, dia))}</span>}
            </button>
          )
        })}
      </div>
    </div>
  )
}

export function SeletorDePeriodo({ valor, onChange, className, disabled, min, max, ariaLabel, inputClassName }: {
  valor: Periodo
  onChange: (p: Periodo) => void
  className?: string
  disabled?: boolean
  /** Limites (ISO) do período inteiro. */
  min?: string
  max?: string
  /** Prefixo dos rótulos acessíveis: "Data de Registro de" / "… até". */
  ariaLabel?: string
  /** Classe extra nos dois <input> (ex.: mf-input-error). */
  inputClassName?: string
}) {
  const [aberto, setAberto] = useState(false)
  // Qual ponta o próximo clique no calendário define.
  const [proximo, setProximo] = useState<'de' | 'ate'>('de')
  const [rascunho, setRascunho] = useState<Periodo>(valor)
  const [previa, setPrevia] = useState<string | null>(null)
  const [esquerda, setEsquerda] = useState<MesVisivel>(() => mesInicial(valor, 2))
  const [posicao, setPosicao] = useState<{ top: number; left: number } | null>(null)
  const [meses, setMeses] = useState(2)
  const raizRef = useRef<HTMLDivElement>(null)
  const painelRef = useRef<HTMLDivElement>(null)

  function abrir(ponta: 'de' | 'ate') {
    if (disabled) return
    const qtd = window.innerWidth < 720 ? 1 : 2
    setMeses(qtd)
    if (!aberto) {
      setRascunho(valor)
      setEsquerda(mesInicial(valor, qtd))
    }
    setProximo(ponta)
    setPrevia(null)
    setAberto(true)
  }
  function fechar() {
    setAberto(false)
    setPrevia(null)
  }

  // Posição fixa abaixo dos campos, sem passar da borda direita da janela.
  useEffect(() => {
    if (!aberto) return
    function atualizar() {
      const r = raizRef.current?.getBoundingClientRect()
      if (!r) return
      const largura = meses === 2 ? LARGURA_DOIS_MESES : LARGURA_UM_MES
      setPosicao({ top: r.bottom + 6, left: Math.max(12, Math.min(r.left, window.innerWidth - largura - 12)) })
    }
    atualizar()
    window.addEventListener('scroll', atualizar, true)
    window.addEventListener('resize', atualizar)
    return () => {
      window.removeEventListener('scroll', atualizar, true)
      window.removeEventListener('resize', atualizar)
    }
  }, [aberto, meses])

  useEffect(() => {
    if (!aberto) return
    function aoClicarFora(e: MouseEvent) {
      const alvo = e.target as Node
      if (raizRef.current?.contains(alvo) || painelRef.current?.contains(alvo)) return
      fechar()
    }
    function aoTeclar(e: KeyboardEvent) { if (e.key === 'Escape') fechar() }
    document.addEventListener('mousedown', aoClicarFora)
    document.addEventListener('keydown', aoTeclar)
    return () => {
      document.removeEventListener('mousedown', aoClicarFora)
      document.removeEventListener('keydown', aoTeclar)
    }
  }, [aberto])

  function escolher(dia: string) {
    if (proximo === 'de' || !rascunho.de || dia < rascunho.de) {
      // Novo início: o fim é escolhido de novo no próximo clique.
      setRascunho({ de: dia, ate: '' })
      setProximo('ate')
    } else {
      setRascunho(r => ({ ...r, ate: dia }))
      setProximo('de')
    }
    setPrevia(null)
  }

  function confirmar() {
    if (!rascunho.de || !rascunho.ate) return
    onChange(rascunho)
    fechar()
  }

  const ativo = aberto ? proximo : null
  const resumo = rascunho.de && rascunho.ate
    ? `${formatar(rascunho.de)} → ${formatar(rascunho.ate)} · ${plural(diasEntre(rascunho.de, rascunho.ate))}`
    : rascunho.de ? `${formatar(rascunho.de)} → escolha a data final` : 'Escolha a data inicial'

  return (
    <div ref={raizRef} className={`mf-periodo${className ? ` ${className}` : ''}`}>
      <span className={`mf-periodo-campo${ativo === 'de' ? ' ativo' : ''}`}>
        <input
          type="date" aria-label={`${ariaLabel ?? 'Período'} de`} value={valor.de} disabled={disabled} className={inputClassName}
          min={min} max={valor.ate || max}
          onClick={() => abrir('de')} onFocus={() => abrir('de')}
          onChange={e => onChange({ ...valor, de: e.target.value })}
        />
        <CalendarBlank size={14} />
      </span>
      <span style={{ color: 'var(--c-neutral)', fontSize: 11 }}>→</span>
      <span className={`mf-periodo-campo${ativo === 'ate' ? ' ativo' : ''}`}>
        <input
          type="date" aria-label={`${ariaLabel ?? 'Período'} até`} value={valor.ate} disabled={disabled} className={inputClassName}
          min={valor.de || min} max={max}
          onClick={() => abrir('ate')} onFocus={() => abrir('ate')}
          onChange={e => onChange({ ...valor, ate: e.target.value })}
        />
        <CalendarBlank size={14} />
      </span>

      {aberto && posicao && createPortal(
        <div ref={painelRef} className="mf-periodo-pop" role="dialog" aria-label="Escolher período" style={{ top: posicao.top, left: posicao.left }}>
          <div className="mf-periodo-meses">
            <button type="button" className="mf-periodo-nav mf-periodo-nav--ant" onClick={() => setEsquerda(e => somarMeses(e, -1))} aria-label="Mês anterior">
              <CaretLeft size={14} weight="bold" />
            </button>
            {Array.from({ length: meses }, (_, i) => (
              <MesDoCalendario
                key={i} visivel={somarMeses(esquerda, i)} rascunho={rascunho}
                previa={proximo === 'ate' ? previa : null}
                onEscolher={escolher} onPassar={setPrevia} min={min} max={max}
              />
            ))}
            <button type="button" className="mf-periodo-nav mf-periodo-nav--prox" onClick={() => setEsquerda(e => somarMeses(e, 1))} aria-label="Próximo mês">
              <CaretRight size={14} weight="bold" />
            </button>
          </div>
          <div className="mf-periodo-rodape">
            <span className="mf-periodo-resumo">{resumo}</span>
            <div style={{ display: 'flex', gap: 8 }}>
              <button type="button" className="mf-btn-clear" onClick={fechar}><X size={14} /> Cancelar</button>
              <button type="button" className="mf-btn-search" onClick={confirmar} disabled={!rascunho.de || !rascunho.ate}>
                <Check size={14} /> Confirmar
              </button>
            </div>
          </div>
        </div>,
        document.body,
      )}
    </div>
  )
}

=== Date picker | SeletorDeData.tsx
import { useEffect, useRef, useState } from 'react'
import { createPortal } from 'react-dom'
import { CalendarBlank, CaretLeft, CaretRight, X } from '@phosphor-icons/react'
import {
  LARGURA_UM_MES, MESES, MesDoCalendario, hojeIso, somarMeses, type MesVisivel,
} from '@/components/shared/SeletorDePeriodo'

// Data única no mesmo visual do SeletorDePeriodo: o campo de data de sempre (aceita digitação), mas
// o clique abre um calendário de um mês. Um clique no dia grava e fecha; o rodapé mostra a data por
// extenso e as opções Limpar e Hoje (as mesmas do calendário nativo). Esc ou clique fora fecham sem
// alterar. Data em ISO (AAAA-MM-DD); "" = vazio.

const DIAS_DA_SEMANA_POR_EXTENSO = ['Domingo', 'Segunda-feira', 'Terça-feira', 'Quarta-feira', 'Quinta-feira', 'Sexta-feira', 'Sábado']

function porExtenso(valor: string): string {
  const [a, m, d] = valor.split('-').map(Number)
  return `${DIAS_DA_SEMANA_POR_EXTENSO[new Date(a, m - 1, d).getDay()]}, ${d} de ${MESES[m - 1].toLowerCase()} de ${a}`
}

function mesDe(valor: string): MesVisivel {
  const [a, m] = (valor || hojeIso()).split('-').map(Number)
  return { ano: a, mes: m - 1 }
}

export function SeletorDeData({
  valor, onChange, className, ariaLabel, id, disabled, readOnly, min, max, inputClassName, compacto, title, onBlur, onEscolher,
}: {
  valor: string
  onChange: (valor: string) => void
  className?: string
  ariaLabel?: string
  id?: string
  disabled?: boolean
  readOnly?: boolean
  /** Limites (ISO): os dias fora deles ficam desabilitados no calendário. */
  min?: string
  max?: string
  /** Classe extra no <input> (ex.: mf-input-error). */
  inputClassName?: string
  /** 28px de altura — campo dentro de linha de tabela. */
  compacto?: boolean
  title?: string
  onBlur?: () => void
  /** Escolha feita no calendário (dia, Limpar ou Hoje) — não dispara enquanto se digita. */
  onEscolher?: (valor: string) => void
}) {
  const [aberto, setAberto] = useState(false)
  const [visivel, setVisivel] = useState<MesVisivel>(() => mesDe(valor))
  const [posicao, setPosicao] = useState<{ top: number; left: number } | null>(null)
  const raizRef = useRef<HTMLDivElement>(null)
  const painelRef = useRef<HTMLDivElement>(null)

  function abrir() {
    if (disabled || readOnly) return
    if (!aberto) setVisivel(mesDe(valor))
    setAberto(true)
  }
  function escolher(dia: string) {
    onChange(dia)
    onEscolher?.(dia)
    setAberto(false)
  }

  // Posição fixa abaixo do campo, sem passar da borda direita da janela.
  useEffect(() => {
    if (!aberto) return
    function atualizar() {
      const r = raizRef.current?.getBoundingClientRect()
      if (!r) return
      setPosicao({ top: r.bottom + 6, left: Math.max(12, Math.min(r.left, window.innerWidth - LARGURA_UM_MES - 12)) })
    }
    atualizar()
    window.addEventListener('scroll', atualizar, true)
    window.addEventListener('resize', atualizar)
    return () => {
      window.removeEventListener('scroll', atualizar, true)
      window.removeEventListener('resize', atualizar)
    }
  }, [aberto])

  useEffect(() => {
    if (!aberto) return
    function aoClicarFora(e: MouseEvent) {
      const alvo = e.target as Node
      if (raizRef.current?.contains(alvo) || painelRef.current?.contains(alvo)) return
      setAberto(false)
    }
    // Esc fecha só o calendário, não o drawer em volta.
    function aoTeclar(e: KeyboardEvent) { if (e.key === 'Escape') { e.stopPropagation(); setAberto(false) } }
    document.addEventListener('mousedown', aoClicarFora)
    document.addEventListener('keydown', aoTeclar, true)
    return () => {
      document.removeEventListener('mousedown', aoClicarFora)
      document.removeEventListener('keydown', aoTeclar, true)
    }
  }, [aberto])

  return (
    <div ref={raizRef} className={`mf-periodo mf-data${compacto ? ' mf-data--compacta' : ''}${className ? ` ${className}` : ''}`}>
      <span className={`mf-periodo-campo${aberto ? ' ativo' : ''}`}>
        <input
          type="date" id={id} aria-label={ariaLabel} title={title} value={valor} className={inputClassName}
          disabled={disabled} readOnly={readOnly} min={min} max={max}
          onClick={abrir} onFocus={abrir} onBlur={onBlur}
          onChange={e => onChange(e.target.value)}
        />
        <CalendarBlank size={14} />
      </span>

      {aberto && posicao && createPortal(
        <div ref={painelRef} className="mf-periodo-pop" role="dialog" aria-label="Escolher data" style={{ top: posicao.top, left: posicao.left }}>
          <div className="mf-periodo-meses">
            <button type="button" className="mf-periodo-nav mf-periodo-nav--ant" onClick={() => setVisivel(v => somarMeses(v, -1))} aria-label="Mês anterior">
              <CaretLeft size={14} weight="bold" />
            </button>
            {/* Início e fim iguais: o dia escolhido aparece destacado, sem faixa de intervalo. */}
            <MesDoCalendario
              visivel={visivel} rascunho={{ de: valor, ate: valor }} previa={null}
              onEscolher={escolher} onPassar={() => {}} min={min} max={max}
            />
            <button type="button" className="mf-periodo-nav mf-periodo-nav--prox" onClick={() => setVisivel(v => somarMeses(v, 1))} aria-label="Próximo mês">
              <CaretRight size={14} weight="bold" />
            </button>
          </div>
          <div className="mf-periodo-rodape mf-data-rodape">
            <span className="mf-periodo-resumo">{valor ? porExtenso(valor) : 'Nenhuma data escolhida'}</span>
            <div className="mf-data-botoes">
              <button type="button" className="mf-btn-clear" onClick={() => escolher('')}><X size={14} /> Limpar</button>
              <button type="button" className="mf-btn-search" onClick={() => escolher(hojeIso())}
                disabled={(!!min && hojeIso() < min) || (!!max && hojeIso() > max)}>Hoje</button>
            </div>
          </div>
        </div>,
        document.body,
      )}
    </div>
  )
}

function addDias(date: Date, n: number, soUteis: boolean): Date {
  const d = new Date(date)
  let added = 0
  const dir = n >= 0 ? 1 : -1
  const target = Math.abs(n)
  while (added < target) {
    d.setDate(d.getDate() + dir)
    if (!soUteis || (d.getDay() !== 0 && d.getDay() !== 6)) added++
  }
  return d
}

function diffDias(a: Date, b: Date, soUteis: boolean): number {
  if (a > b) [a, b] = [b, a]
  if (!soUteis) {
    return Math.round((b.getTime() - a.getTime()) / 86_400_000)
  }
  let count = 0
  const d = new Date(a)
  while (d < b) {
    d.setDate(d.getDate() + 1)
    if (d.getDay() !== 0 && d.getDay() !== 6) count++
  }
  return count
}

    const d = diffDias(new Date(dateA + 'T12:00:00'), new Date(dateB + 'T12:00:00'), soUteisEntre)
    setResultEntre(`${d} dia${d !== 1 ? 's' : ''} ${soUteisEntre ? 'úteis' : 'corridos'}`)
  }

  function calcSemana() {
    if (!dateSemana) return
    const d = new Date(dateSemana + 'T12:00:00')
    setResultSemana(`${DIAS[d.getDay()]} — ${fmt(d)}`)
  }

  const abaClass = (a: Aba) => `calc-aba${aba === a ? ' is-ativa' : ''}`

  return (
    <div className="modal-ov">
      <div className="modal calc-datas-modal">
        <div className="modal__hdr">
          <h2 className="modal__titulo">Utilitário de Datas</h2>
          <button onClick={onClose} className="modal__fechar" aria-label="Fechar"><X size={16} /></button>
        </div>

        <div className="tabs">
          <button className={abaClass('calc')} onClick={() => setAba('calc')}>Cálculo</button>
          <button className={abaClass('entre')} onClick={() => setAba('entre')}>Entre datas</button>
          <button className={abaClass('diaSemana')} onClick={() => setAba('diaSemana')}>Dia da semana</button>
        </div>

        <div className="calc-corpo">
          {aba === 'calc' && (
            <>
              <label className="fg"><span>Data base</span>
                <SeletorDeData valor={baseDate} onChange={setBaseDate} ariaLabel="Data base" />
              </label>
              <label className="fg"><span>Número de dias (negativo para subtrair)</span>
                <input type="number" value={dias} onChange={e => setDias(e.target.value)} />
              </label>
              <label className="check-lbl">
                <input type="checkbox" checked={soUteis} onChange={e => setSoUteis(e.target.checked)} />
                Contar apenas dias úteis (seg–sex)
              </label>
              <button onClick={calcDatas} className="btn btn--success btn--filled btn--medium">Calcular</button>
              {resultCalc && <p className="calc-resultado">{resultCalc}</p>}
            </>
          )}

          {aba === 'entre' && (
            <>
              <label className="fg"><span>Data inicial</span>
                <SeletorDeData valor={dateA} onChange={setDateA} ariaLabel="Data inicial" />
              </label>
              <label className="fg"><span>Data final</span>
                <SeletorDeData valor={dateB} onChange={setDateB} ariaLabel="Data final" />
              </label>
              <label className="check-lbl">
                <input type="checkbox" checked={soUteisEntre} onChange={e => setSoUteisEntre(e.target.checked)} />
                Contar apenas dias úteis
              </label>
              <button onClick={calcEntre} className="btn btn--success btn--filled btn--medium">Calcular</button>
              {resultEntre && <p className="calc-resultado">{resultEntre}</p>}
            </>
          )}

          {aba === 'diaSemana' && (
            <>
              <label className="fg"><span>Data</span>
                <SeletorDeData valor={dateSemana} onChange={setDateSemana} ariaLabel="Data" />
              </label>
              <button onClick={calcSemana} className="btn btn--success btn--filled btn--medium">Verificar</button>
              {resultSemana && <p className="calc-resultado">{resultSemana}</p>}
            </>
          )}
        </div>
      </div>
    </div>
  )
}
